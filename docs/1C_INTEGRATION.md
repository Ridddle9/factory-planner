# Интеграция с 1С — внешняя обработка

Внешняя обработка (`.epf`) для выгрузки заказов из 1С в планировщик по кнопке.

---

## Как создать обработку в конфигураторе

1. Файл → Новый → Внешняя обработка
2. Имя: `ВыгрузкаВПланировщик`, Синоним: `Выгрузка в планировщик`
3. Перейти в **Формы** → добавить форму (тип — Форма обработки)
4. На форме разместить элементы (см. ниже)
5. Вставить код из раздела «Модуль формы»
6. Сохранить как `.epf`

---

## Элементы формы

| Имя элемента     | Тип          | Заголовок                  | Значение по умолчанию |
|------------------|--------------|----------------------------|-----------------------|
| `СерверАдрес`    | Поле ввода   | Адрес сервера              | `192.168.1.100`       |
| `СерверПорт`     | Поле ввода   | Порт                       | `3000`                |
| `РезультатТекст` | Текстовое поле (многострочное, только чтение) | Результат | —   |
| `КнопкаВыгрузить`| Кнопка       | Выгрузить в планировщик    | —                     |

Привязки реквизитов формы:
- `СерверАдрес` → реквизит формы `СерверАдрес` (Строка)
- `СерверПорт` → реквизит формы `СерверПорт` (Число)

---

## Модуль формы

```bsl
// ============================================================
// Реквизиты формы (создать через конфигуратор):
//   СерверАдрес  — Строка
//   СерверПорт   — Число
// ============================================================

&НаКлиенте
Процедура КнопкаВыгрузитьНажатие(Элемент)
    Если ПустаяСтрока(СерверАдрес) Тогда
        Предупреждение("Укажите адрес сервера");
        Возврат;
    КонецЕсли;

    РезультатТекст = "Выгружаю данные...";

    Результат = ВыгрузитьНаСервер(СерверАдрес, СерверПорт);
    РезультатТекст = Результат;
КонецПроцедуры

&НаСервереБезКонтекста
Функция ВыгрузитьНаСервер(Адрес, Порт)
    Попытка
        Данные = СобратьДанные();

        Если Данные.Количество() = 0 Тогда
            Возврат "Нет позиций к обеспечению (К обеспечению = 0 у всех заказов)";
        КонецЕсли;

        ТелоJSON = СформироватьJSON(Данные);
        Ответ = ОтправитьПОСТ(Адрес, Порт, "/import/1c", ТелоJSON);
        Возврат РасшифроватьОтвет(Ответ, Данные.Количество());
    Исключение
        Возврат "Ошибка: " + ОписаниеОшибки();
    КонецПопытки;
КонецФункции

// ------------------------------------------------------------
// Запрос данных из 1С
// ------------------------------------------------------------

&НаСервереБезКонтекста
Функция СобратьДанные()
    // Запрос рассчитан на 1С:УТ 11 / ERP.
    // Если конфигурация другая — адаптируйте имена регистра и полей.
    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ
    |    Ном.ВидНоменклатуры.Наименование КАК ВидНоменклатуры,
    |    Ном.Код                           КАК Код,
    |    Ном.Наименование                  КАК Номенклатура,
    |    Зак.Ссылка.Контрагент             КАК Клиент,
    |    Зак.Ссылка.Номер                  КАК НомерЗаказа,
    |    Зак.Ссылка.ГрафикОплаты           КАК ГрафикОплаты,
    |    Зак.Ссылка.Менеджер               КАК Менеджер,
    |    Зак.Ссылка.Дата                   КАК ДатаЗаказа,
    |    Зак.Ссылка.ДатаСогласования       КАК ДатаСогласования,
    |    Зак.КоличествоОстаток             КАК КОбеспечению
    |ИЗ
    |    РегистрНакопления.ЗаказыКлиентов.Остатки() КАК Зак
    |        ЛЕВОЕ СОЕДИНЕНИЕ Справочник.Номенклатура КАК Ном
    |            ПО Зак.Номенклатура = Ном.Ссылка
    |ГДЕ
    |    Зак.КоличествоОстаток > 0
    |УПОРЯДОЧИТЬ ПО
    |    Зак.Ссылка.Дата";

    Результат = Запрос.Выполнить();
    Возврат Результат.Выгрузить();
КонецФункции

// ------------------------------------------------------------
// Формирование JSON
// ------------------------------------------------------------

&НаСервереБезКонтекста
Функция СформироватьJSON(ТаблицаДанных)
    ЗаписьJSON = Новый ЗаписьJSON;
    НастройкиJSON = Новый НастройкиСериализацииJSON;
    НастройкиJSON.ФорматДатыJSON = ФорматДатыJSON.ISO;

    ПотокJSON = Новый ЗаписьJSON;
    ПотокJSON.УстановитьСтроку(Новый НастройкиЗаписиJSON(Ложь, " "));
    ПотокJSON.ЗаписатьНачалоМассива();

    Для Каждого Строка Из ТаблицаДанных Цикл
        ПотокJSON.ЗаписатьНачалоОбъекта();

        ЗаписатьПолеJSON(ПотокJSON, "categoryName",  Строка.ВидНоменклатуры);
        ЗаписатьПолеJSON(ПотокJSON, "productCode",   СокрЛП(Строка.Код));
        ЗаписатьПолеJSON(ПотокJSON, "productName",   СокрЛП(Строка.Номенклатура));
        ЗаписатьПолеJSON(ПотокJSON, "customer",      Строка(Строка.Клиент));
        ЗаписатьПолеJSON(ПотокJSON, "orderNumber",   СокрЛП(Строка.НомерЗаказа));
        ЗаписатьПолеJSON(ПотокJSON, "paymentTerms",  Строка(Строка.ГрафикОплаты));
        ЗаписатьПолеJSON(ПотокJSON, "manager",       Строка(Строка.Менеджер));
        ЗаписатьПолеJSON(ПотокJSON, "orderDate",     ФорматДаты(Строка.ДатаЗаказа));
        ЗаписатьПолеJSON(ПотокJSON, "approvedAt",    ФорматДаты(Строка.ДатаСогласования));

        ПотокJSON.ЗаписатьИмяСвойства("remaining");
        ПотокJSON.ЗаписатьЗначение(Строка.КОбеспечению);

        ПотокJSON.ЗаписатьКонецОбъекта();
    КонецЦикла;

    ПотокJSON.ЗаписатьКонецМассива();
    Возврат ПотокJSON.Закрыть();
КонецФункции

&НаСервереБезКонтекста
Процедура ЗаписатьПолеJSON(Поток, Имя, Значение)
    Поток.ЗаписатьИмяСвойства(Имя);
    Поток.ЗаписатьЗначение(СокрЛП(Строка(Значение)));
КонецПроцедуры

&НаСервереБезКонтекста
Функция ФорматДаты(Дата)
    Если Дата = '00010101' Или НЕ ЗначениеЗаполнено(Дата) Тогда
        Возврат "";
    КонецЕсли;
    Возврат Формат(Дата, "ДФ=dd.MM.yyyy");
КонецФункции

// ------------------------------------------------------------
// HTTP-запрос
// ------------------------------------------------------------

&НаСервереБезКонтекста
Функция ОтправитьПОСТ(Адрес, Порт, Путь, Тело)
    Соединение = Новый HTTPСоединение(
        Адрес,
        Порт,
        ,         // пользователь
        ,         // пароль
        ,         // прокси
        30        // таймаут (сек)
    );

    Запрос = Новый HTTPЗапрос(Путь);
    Запрос.Заголовки.Вставить("Content-Type", "application/json; charset=utf-8");
    Запрос.УстановитьТелоИзСтроки(Тело, КодировкаТекста.UTF8);

    Возврат Соединение.ОтправитьДляОбработки(Запрос);
КонецФункции

// ------------------------------------------------------------
// Разбор ответа
// ------------------------------------------------------------

&НаСервереБезКонтекста
Функция РасшифроватьОтвет(Ответ, КолвоСтрок)
    КодОтвета = Ответ.КодСостояния;
    ТелоОтвета = Ответ.ПолучитьТелоКакСтроку(КодировкаТекста.UTF8);

    Если КодОтвета <> 200 Тогда
        Возврат "Сервер вернул ошибку " + КодОтвета + ":" + Символы.ПС + ТелоОтвета;
    КонецЕсли;

    Попытка
        ЧтениеJSON = Новый ЧтениеJSON;
        ЧтениеJSON.УстановитьСтроку(ТелоОтвета);
        Данные = ПрочитатьJSON(ЧтениеJSON);
    Исключение
        Возврат "Ответ получен, но не удалось разобрать JSON:" + Символы.ПС + ТелоОтвета;
    КонецПопытки;

    Итог = Данные["summary"];
    Конфликты = Данные["conflicts"];
    Факты = Данные["factDetails"];

    Текст = "✓ Выгружено строк из 1С: " + КолвоСтрок + Символы.ПС + Символы.ПС;

    Текст = Текст + "Категории:    создано " + Итог["categories"]["created"]
                  + ", обновлено " + Итог["categories"]["updated"] + Символы.ПС;
    Текст = Текст + "Товары:       создано " + Итог["products"]["created"]
                  + ", обновлено " + Итог["products"]["updated"] + Символы.ПС;
    Текст = Текст + "Заказы:       создано " + Итог["orders"]["created"]
                  + ", обновлено " + Итог["orders"]["updated"] + Символы.ПС;
    Текст = Текст + "Позиции:      создано " + Итог["orderItems"]["created"]
                  + ", обновлено " + Итог["orderItems"]["updated"] + Символы.ПС;
    Текст = Текст + "Факт записан: " + Итог["factRecorded"] + " позиций" + Символы.ПС;

    Если Конфликты <> Неопределено И Конфликты.Количество() > 0 Тогда
        Текст = Текст + Символы.ПС + "⚠ Конфликты факта (" + Конфликты.Количество() + "):" + Символы.ПС;
        Для Каждого К Из Конфликты Цикл
            Текст = Текст + "  Заказ " + К["orderNumber"] + ", товар " + К["productCode"]
                  + ": в системе " + К["existingFact"] + ", из 1С " + К["calculatedFact"] + Символы.ПС;
        КонецЦикла;
        Текст = Текст + "→ Разрешите конфликты вручную в разделе «Импорт Excel» планировщика";
    КонецЕсли;

    Возврат Текст;
КонецФункции
```

## Адаптация под вашу конфигурацию

Запрос написан для **1С:УТ 11 / ERP**. Если конфигурация другая:

| Конфигурация | Что поменять в запросе |
|---|---|
| УТ 10.3 / УПП | Регистр: `ЗаказыПокупателей`, поле: `КоличествоОстаток` |
| Самописная | Замените регистр и поля на свои; `КОбеспечению` — то, что уже выгружается в Excel |

Проще всего: откройте текущий отчёт (который выгружает Excel), скопируйте его запрос и вставьте вместо `Запрос.Текст`, добавив фильтр `КОбеспечению > 0`.

## Итоговый flow после внедрения

```
1С (кнопка "Выгрузить") → POST /import/1c → планировщик
```

Вместо: **1С → Excel → браузер → "Импорт Excel"**
